# T-601 Web App Shell + Design System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create `apps/web` (`@bombe/web`) — a Next.js 16 App Router app with Tailwind v4, implementing the DESIGN.md token system, a base component library, app shell (nav + layout + footer), SSE infrastructure, a polished landing page, and deterministic tests — without breaking `pnpm run ci`.

**Architecture:** Next.js 16 App Router with Tailwind v4 CSS-variable `@theme`. The app is fully typed (TS5 strict). It imports `@bombe/shared` for `SseEventSchema` / `SseEvent`. A server-side `GET /api/stream` route emits a deterministic SSE demo sequence; a client-side hook `useEventStream` consumes it. `apps/web` has its OWN `tsconfig.json` (using Next.js types + `"jsx":"preserve"`) and is **excluded** from the root `tsconfig.json` to prevent interference. The root `ci` script grows one tail: `&& pnpm --filter @bombe/web typecheck`.

**Tech Stack:** Next.js 16, React 19, Tailwind v4 (CSS-first), TypeScript 5 strict, Vitest + @testing-library/react, `@bombe/shared`, Google Fonts (Inter), next/font/google.

```typescript
"use client";

import { useState } from "react";

interface MonoProps {
  value: string;
  truncate?: boolean;
  truncateChars?: number;
  showCopy?: boolean;
  className?: string;
}

function truncateHash(value: string, chars: number): string {
  if (value.length <= chars * 2 + 3) return value;
  return `${value.slice(0, chars)}…${value.slice(-chars)}`;
}

export function Mono({
  value,
  truncate = false,
  truncateChars = 6,
  showCopy = true,
  className = "",
}: MonoProps) {
  const [copied, setCopied] = useState(false);

  const display = truncate ? truncateHash(value, truncateChars) : value;

  function handleCopy() {
    void navigator.clipboard.writeText(value).then(() => {
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    });
  }

  return (
    <span className={`inline-flex items-center gap-1.5 ${className}`}>
      <code
        className="font-mono text-[13px] text-[#8d969e] bg-[#16181a] px-2 py-0.5 rounded-[8px]"
        title={value}
      >
        {display}
      </code>
      {showCopy && (
        <button
          type="button"
          onClick={handleCopy}
          className="text-[#505a63] hover:text-[#ffffff] transition-colors cursor-pointer text-[11px]"
          aria-label="Copy to clipboard"
        >
          {copied ? "✓" : "⧉"}
        </button>
      )}
    </span>
  );
}
```

```typescript
import type { ReactNode } from "react";

// Explorer link helper: in mock mode renders address/tx as monospace with
// "mock chain" tooltip (disabled link). In live mode, links to Mantle Sepolia
// explorer (chain 5003). Mode is set at boot via NEXT_PUBLIC_APP_MODE.

const MANTLE_SEPOLIA_EXPLORER = "https://sepolia.mantlescan.xyz";

interface ExplorerLinkProps {
  type: "address" | "tx";
  value: string;
  children?: ReactNode;
  className?: string;
}

function truncate(value: string, chars = 6): string {
  if (value.length <= chars * 2 + 3) return value;
  return `${value.slice(0, chars)}…${value.slice(-chars)}`;
}

export function ExplorerLink({ type, value, children, className = "" }: ExplorerLinkProps) {
  // NEXT_PUBLIC_APP_MODE is set at build time; default to mock
  const isMock =
    typeof process !== "undefined"
      ? (process.env["NEXT_PUBLIC_APP_MODE"] ?? "mock") === "mock"
      : true;

  const display = children ?? (
    <code className="font-mono text-[13px] text-[#8d969e]">{truncate(value)}</code>
  );

  if (isMock) {
    return (
      <span
        className={`inline-flex items-center gap-1 group relative cursor-default ${className}`}
        title="Links disabled in mock mode — no live chain"
      >
        {display}
        <span className="text-[11px] text-[#505a63] border border-[#505a63] rounded px-1">
          mock
        </span>
      </span>
    );
  }

  const href =
    type === "address"
      ? `${MANTLE_SEPOLIA_EXPLORER}/address/${value}`
      : `${MANTLE_SEPOLIA_EXPLORER}/tx/${value}`;

  return (
    <a
      href={href}
      target="_blank"
      rel="noopener noreferrer"
      className={`inline-flex items-center gap-1 text-[#376cd5] hover:underline ${className}`}
```

```typescript
import { SseEventSchema } from "@bombe/shared";
import type { SseEvent } from "@bombe/shared";

export type ParseResult =
  | { ok: true; event: SseEvent }
  | { ok: false; error: string };

/**
 * Pure function: parse a raw SSE data string into a typed SseEvent.
 * Extracted from useEventStream so it's testable without a real EventSource.
 */
export function parseEvent(data: string): ParseResult {
  let raw: unknown;
  try {
    raw = JSON.parse(data);
  } catch {
    return { ok: false, error: `JSON parse failed: ${data.slice(0, 100)}` };
  }

  const result = SseEventSchema.safeParse(raw);
  if (!result.success) {
    return {
      ok: false,
      error: `Schema validation failed: ${result.error.message}`,
    };
  }
  return { ok: true, event: result.data };
}
```

```typescript
"use client";

import { useEffect, useReducer, useRef } from "react";
import type { SseEvent, ClaimPostedEvent, AgentDoneEvent, AgentStepEvent } from "@bombe/shared";
import { parseEvent } from "./parseEvent";

export interface StreamState {
  connected: boolean;
  events: SseEvent[];
  claims: ClaimPostedEvent[];
  agentDone: AgentDoneEvent[];
  agentSteps: AgentStepEvent[];
  error: string | null;
}

type StreamAction =
  | { type: "connected" }
  | { type: "event"; event: SseEvent }
  | { type: "error"; error: string }
  | { type: "disconnected" };

function reducer(state: StreamState, action: StreamAction): StreamState {
  switch (action.type) {
    case "connected":
      return { ...state, connected: true, error: null };
    case "disconnected":
      return { ...state, connected: false };
    case "error":
      return { ...state, error: action.error, connected: false };
    case "event": {
      const { event } = action;
      const next: StreamState = {
        ...state,
        events: [...state.events, event],
      };
      if (event.kind === "CLAIM_POSTED") {
        next.claims = [...state.claims, event];
      } else if (event.kind === "AGENT_DONE") {
        next.agentDone = [...state.agentDone, event];
      } else if (event.kind === "AGENT_STEP") {
        next.agentSteps = [...state.agentSteps, event];
      }
      return next;
    }
    default:
      return state;
  }
}

const initialState: StreamState = {
  connected: false,
  events: [],
  claims: [],
  agentDone: [],
  agentSteps: [],
  error: null,
};

/**
 * Client hook: connects to /api/stream via EventSource, parses SseEvent
 * objects, and routes by kind into typed state buckets.
 */
export function useEventStream(url = "/api/stream"): StreamState {
  const [state, dispatch] = useReducer(reducer, initialState);
  const esRef = useRef<EventSource | null>(null);

  useEffect(() => {
    const es = new EventSource(url);
    esRef.current = es;

    es.onopen = () => dispatch({ type: "connected" });

    es.onmessage = (e: MessageEvent<string>) => {
      const result = parseEvent(e.data);
      if (result.ok) {
        dispatch({ type: "event", event: result.event });
      } else {
        console.warn("[useEventStream] parse error:", result.error);
      }
    };

    es.onerror = () => {
      dispatch({ type: "error", error: "SSE connection error" });
      es.close();
    };

    return () => {
      es.close();
      esRef.current = null;
      dispatch({ type: "disconnected" });
    };
  }, [url]);

  return state;
}
```

## Known tricky bits

1. **Next.js version**: "Next.js 16" in the spec — actual latest package is ^15.x. Use the latest available; the App Router patterns are identical. Check `npm info next dist-tags` first.
2. **verbatimModuleSyntax**: Must be `false` in apps/web tsconfig because Next.js's auto-generated `next-env.d.ts` uses `/// <reference>` directives that conflict with verbatimModuleSyntax.
3. **tailwindcss v4**: Uses `@import "tailwindcss"` + `@theme {}` CSS-first approach (not a JS config file). The postcss plugin is `@tailwindcss/postcss`, not `tailwindcss`.
4. **`@bombe/shared` import path**: shared exports use `.js` extensions internally but Next.js with `transpilePackages` resolves them via the TS path. The `workspace:*` dep + `transpilePackages` in `next.config.ts` handles this.
5. **Root vitest.config.ts** already includes `apps/**/*.test.ts` — it will also pick up web tests during `pnpm test:unit`. That's intentional for the root CI gate. The web package's own `vitest.config.ts` is needed for `pnpm --filter @bombe/web test` to work with jsdom + React plugin.
